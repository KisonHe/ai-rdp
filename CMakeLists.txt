cmake_minimum_required(VERSION 3.22)
project(ai_rdp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RDP REQUIRED IMPORTED_TARGET freerdp3 winpr3)
find_package(OpenSSL REQUIRED)
add_executable(ai-rdp-native native/main.cpp)
target_link_libraries(ai-rdp-native PRIVATE PkgConfig::RDP OpenSSL::Crypto)
target_compile_options(ai-rdp-native PRIVATE -Wall -Wextra -Wpedantic)
