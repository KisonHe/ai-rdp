cmake_minimum_required(VERSION 3.22)
project(ai_rdp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RDP REQUIRED IMPORTED_TARGET freerdp3 winpr3)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(ai-rdp-native native/main.cpp)
target_link_libraries(ai-rdp-native PRIVATE PkgConfig::RDP OpenSSL::Crypto Threads::Threads)
if(WIN32)
  target_compile_definitions(ai-rdp-native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(ai-rdp-native PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(ai-rdp-native PRIVATE /W4 /permissive-)
else()
  target_compile_options(ai-rdp-native PRIVATE -Wall -Wextra -Wpedantic)
endif()
